from flask import Flask, render_template, request, redirect, url_for, session, flash from flask_mysqldb import MySQL import MySQLdb.cursors import re import pickle import numpy as np from ml.model import classify_case from config import Config app = Flask(__name__) app.secret_key = Config.SECRET_KEY app.config.from_object(Config) mysql = MySQL(app) # Load pre-trained ML model with open('models/classifier.pkl', 'rb') as model_file: model = pickle.load(model_file) @app.route('/') def index(): return redirect(url_for('login')) @app.route('/login', methods=['GET', 'POST']) def login(): if request.method == 'POST': username = request.form['username'] password = request.form['password'] cursor = mysql.connection.cursor(MySQLdb.cursors.DictCursor) cursor.execute('SELECT * FROM users WHERE username = %s AND password = %s', (username, password)) account = cursor.fetchone() if account: session['loggedin'] = True session['id'] = account['id'] session['username'] = account['username'] return redirect(url_for('dashboard')) else: flash('Incorrect username/password!', 'danger') return render_template('login.html') @app.route('/register', methods=['GET', 'POST']) def register(): if request.method == 'POST': username = request.form['username'] password = request.form['password'] cursor = mysql.connection.cursor() cursor.execute('INSERT INTO users (username, password) VALUES (%s, %s)', (username, password)) mysql.connection.commit() flash('You have successfully registered!', 'success') return redirect(url_for('login')) return render_template('register.html') @app.route('/dashboard') def dashboard(): if 'loggedin' in session: cursor = mysql.connection.cursor(MySQLdb.cursors.DictCursor) cursor.execute('SELECT * FROM cases') cases = cursor.fetchall() return render_template('dashboard.html', cases=cases) return redirect(url_for('login')) @app.route('/add_case', methods=['GET', 'POST']) def add_case(): if 'loggedin' in session: if request.method == 'POST': title = request.form['title'] description = request.form['description'] urgency = int(request.form['urgency']) complexity = int(request.form['complexity']) inputs = np.array([[urgency, complexity]]) category = classify_case(inputs, model) cursor = mysql.connection.cursor() cursor.execute('INSERT INTO cases (title, description, urgency, complexity, category) render_template('add_case.html') return redirect(url_for('login')) @app.route('/schedule') def schedule(): if 'loggedin' in session: cursor = mysql.connection.cursor(MySQLdb.cursors.DictCursor) redirect(url_for('login')) @app.route('/logout') def logout(): session.pop('loggedin', None) session.pop('id', None) session.pop('username', None) flash('You have been logged out!', 'info') return
